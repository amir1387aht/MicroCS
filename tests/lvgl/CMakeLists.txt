cmake_minimum_required(VERSION 3.16)
project(microcs_lvgl_tests C CXX ASM)
if(NOT LVGL_HOST_DIR)
    message(FATAL_ERROR "Pass -DLVGL_HOST_DIR=/path/to/lvgl-8.x-or-9.x (not downloaded automatically)")
endif()
option(NO_COMPRESSED_FONTS "Test rejection of binary fonts requiring disabled host decompression" OFF)
if(NO_COMPRESSED_FONTS)
    add_compile_definitions(MCS_LVGL_TEST_NO_COMPRESSED=1)
endif()
option(MINIMAL_WIDGETS "Disable exposed optional widgets" OFF)
if(MINIMAL_WIDGETS)
    add_compile_definitions(MCS_LVGL_TEST_MINIMAL=1)
endif()
option(SANITIZE "ASan/UBSan all libraries and tests" OFF)
if(SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
# Early LVGL 8.0 has no standalone target in its CMake file. The TEST host
# builds its own dependency (the production wrapper never compiles LVGL).
if(EXISTS "${LVGL_HOST_DIR}/env_support/cmake/custom.cmake" OR EXISTS "${LVGL_HOST_DIR}/env_support/cmake/main.cmake")
    add_subdirectory(${LVGL_HOST_DIR} host-lvgl)
else()
    file(GLOB_RECURSE host_sources "${LVGL_HOST_DIR}/src/*.c")
    add_library(lvgl STATIC ${host_sources})
    add_library(lvgl::lvgl ALIAS lvgl)
    target_include_directories(lvgl PUBLIC ${LVGL_HOST_DIR} ${CMAKE_CURRENT_LIST_DIR})
    target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
endif()
set(MICROCS_LVGL ON CACHE BOOL "" FORCE)
set(MICROCS_MODULES ON CACHE BOOL "" FORCE)
set(MICROCS_CLI OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. microcs)
set(_test_font "")
foreach(path tests/src/test_assets/test_font_1.fnt tests/src/test_assets/font_1.fnt tests/src/test_fonts/font_1.fnt tests/font_1.fnt)
    if(EXISTS "${LVGL_HOST_DIR}/${path}")
        set(_test_font "${LVGL_HOST_DIR}/${path}")
        break()
    endif()
endforeach()
if(NOT _test_font)
    message(FATAL_ERROR "Asset tests require an upstream binary font fixture; no fixture was silently skipped")
endif()
add_executable(test_lvgl test_lvgl.c test_assets.c test_native.c)
target_compile_definitions(test_lvgl PRIVATE MCS_TEST_FONT_FILE="${_test_font}" MCS_LVGL_EXAMPLE_DIR="${CMAKE_CURRENT_LIST_DIR}/../../examples/lvgl")
find_package(Threads REQUIRED)
target_link_libraries(test_lvgl PRIVATE microcs m Threads::Threads)
target_compile_options(test_lvgl PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME host_owned_lvgl COMMAND test_lvgl)
set_tests_properties(host_owned_lvgl PROPERTIES TIMEOUT 120)
