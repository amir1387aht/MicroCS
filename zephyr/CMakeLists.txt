# MicroCS as a Zephyr module: CONFIG_MICROCS=y in prj.conf
if(CONFIG_MICROCS)
    set(MICROCS_DIR ${CMAKE_CURRENT_LIST_DIR}/..)
    file(GLOB MICROCS_CORE_SRCS ${MICROCS_DIR}/src/*.c)
    file(GLOB MICROCS_MODULE_SRCS ${MICROCS_DIR}/modules/*/*.c)
    list(FILTER MICROCS_MODULE_SRCS EXCLUDE REGEX "mcs_vfs_posix\\.c$")   # host-only backend
    zephyr_library_named(microcs)
    zephyr_library_sources(${MICROCS_CORE_SRCS} ${MICROCS_MODULE_SRCS})
    zephyr_library_sources_ifdef(CONFIG_MICROCS_PORT ${MICROCS_DIR}/ports/zephyr/mcs_port_zephyr.c)
    zephyr_include_directories(${MICROCS_DIR}/include ${MICROCS_DIR}/ports/zephyr)
    # Kconfig profile = the default; MCS_PROFILE in mcs_user_config.h wins over it
    foreach(p AUTO EMBEDDED MCU LOWRAM MIN)
        if(CONFIG_MICROCS_PROFILE_${p})
            zephyr_compile_definitions(MCS_DEFAULT_PROFILE=MCS_PROFILE_${p})
        endif()
    endforeach()
    # project config header: -DMICROCS_CONFIG=..., CONFIG_MICROCS_USER_CONFIG_FILE, or
    # mcs_user_config.h in the application folder, its include/ or src/
    if(NOT DEFINED MICROCS_CONFIG)
        set(MICROCS_CONFIG "${CONFIG_MICROCS_USER_CONFIG_FILE}")
    endif()
    set(_mcs_cfg "")
    if(MICROCS_CONFIG STREQUAL "NONE")
        zephyr_compile_definitions(MCS_USER_CONFIG=0)
    elseif(MICROCS_CONFIG)
        set(_mcs_cfg "${MICROCS_CONFIG}")
        if(NOT IS_ABSOLUTE "${_mcs_cfg}" AND EXISTS "${APPLICATION_SOURCE_DIR}/${_mcs_cfg}")
            get_filename_component(_mcs_cfg "${APPLICATION_SOURCE_DIR}/${_mcs_cfg}" ABSOLUTE)
        endif()
    else()
        foreach(d "" "include/" "src/")
            if(NOT _mcs_cfg AND EXISTS "${APPLICATION_SOURCE_DIR}/${d}mcs_user_config.h")
                set(_mcs_cfg "${APPLICATION_SOURCE_DIR}/${d}mcs_user_config.h")
            endif()
        endforeach()
    endif()
    if(_mcs_cfg)
        message(STATUS "MicroCS: config header ${_mcs_cfg}")
        zephyr_compile_definitions(MCS_USER_CONFIG_FILE="${_mcs_cfg}")
    endif()
    if(CONFIG_MICROCS_PORT)
        zephyr_compile_definitions(MCS_PORT_HAL=1)
    endif()
    if(NOT CONFIG_MICROCS_WS2812)
        zephyr_compile_definitions(MCS_ENABLE_WS2812=0)
    endif()
    if(CONFIG_FILE_SYSTEM_LITTLEFS)
        zephyr_library_link_libraries(LITTLEFS)    # lfs.h for the /lfs mount helper
    endif()
    if(CONFIG_MICROCS_YAFFS2)
        include(${MICROCS_DIR}/cmake/MicroCSFS.cmake)
        microcs_yaffs2_sources()
        zephyr_library_sources(${MICROCS_YAFFS_SRCS})
        zephyr_include_directories(${MICROCS_YAFFS_INC})
        zephyr_compile_definitions(MCS_ENABLE_YAFFS=1 MCS_YAFFS_OSGLUE=1 ${MICROCS_YAFFS_DEFS})
        set_source_files_properties(${MICROCS_YAFFS_SRCS} PROPERTIES COMPILE_OPTIONS "-w;-include;sys/types.h")
    endif()
    zephyr_library_compile_options(-Wno-unused-parameter)
endif()
