# MicroCS as a Zephyr module: CONFIG_MICROCS=y in prj.conf
if(CONFIG_MICROCS)
    set(MICROCS_DIR ${CMAKE_CURRENT_LIST_DIR}/..)
    file(GLOB MICROCS_CORE_SRCS ${MICROCS_DIR}/src/*.c)
    file(GLOB MICROCS_MODULE_SRCS ${MICROCS_DIR}/modules/*/*.c)
    zephyr_library_named(microcs)
    zephyr_library_sources(${MICROCS_CORE_SRCS} ${MICROCS_MODULE_SRCS})
    zephyr_library_sources_ifdef(CONFIG_MICROCS_PORT ${MICROCS_DIR}/ports/zephyr/mcs_port_zephyr.c)
    zephyr_include_directories(${MICROCS_DIR}/include ${MICROCS_DIR}/ports/zephyr)
    if(CONFIG_MICROCS_PROFILE_LOWRAM)
        zephyr_compile_definitions(MCS_USER_CONFIG_FILE="profiles/mcs_profile_lowram.h")
    endif()
    zephyr_library_compile_options(-Wno-unused-parameter)
endif()
